from jinja2 import Template
import pandas as pd

# Simulated LLM output
title = "Batch Job Summary - October"
message = "The following jobs were processed successfully:"
data = [
    {"Job": "JOB001", "Status": "Success", "Time": "2 mins"},
    {"Job": "JOB002", "Status": "Failed", "Time": "1 min"},
]

# Convert dict data to HTML table
df = pd.DataFrame(data)
table_html = df.to_html(index=False, border=0, classes="data-table")

# Load HTML template
with open("email_template.html") as f:
    template = Template(f.read())

# Render HTML
html_body = template.render(title=title, message=message, table_html=table_html)

import win32com.client as win32

outlook = win32.Dispatch("outlook.application")
mail = outlook.CreateItem(0)
mail.To = "recipient@company.com"
mail.Subject = "Automated Job Summary"
mail.HTMLBody = html_body
mail.Send()

print("✅ HTML email sent via Outlook!")

import smtplib
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText

msg = MIMEMultipart("alternative")
msg["From"] = "your_email@company.com"
msg["To"] = "recipient@company.com"
msg["Subject"] = "Automated Job Summary"

msg.attach(MIMEText(html_body, "html"))

server = smtplib.SMTP("smtp.office365.com", 587)
server.starttls()
server.login("your_email@company.com", "your_password_or_app_password")
server.send_message(msg)
server.quit()

print("✅ HTML email sent via SMTP!")
